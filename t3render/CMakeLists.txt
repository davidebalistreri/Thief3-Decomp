function(t3render_add_scene_pbr_d3d12)
  add_executable(t3render_scene_pbr_d3d12 src/scene_pbr_d3d12.cpp)
  target_compile_features(t3render_scene_pbr_d3d12 PRIVATE cxx_std_17)
  target_compile_definitions(t3render_scene_pbr_d3d12 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(t3render_scene_pbr_d3d12 PRIVATE d3d12 dxgi dxguid)
  if(MSVC)
    target_compile_options(t3render_scene_pbr_d3d12 PRIVATE /W4 /utf-8)
    set_property(TARGET t3render_scene_pbr_d3d12 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(t3render_scene_pbr_d3d12 PRIVATE -Wall -Wextra)
    target_link_options(t3render_scene_pbr_d3d12 PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endfunction()

include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <windows.h>\n#include <d3d8.h>\nint main() { return 0; }" T3RENDER_HAS_D3D8_H)
if(NOT T3RENDER_HAS_D3D8_H)
  message(FATAL_ERROR "The Direct3D 8 SDK header d3d8.h is required in the compiler include path.")
endif()

# Portable utilities used by the optional D3D8 buffer observer.
add_library(t3render_ir STATIC ir/buffer_shadow.cpp ir/buffer_registry.cpp
  ir/vertex_declaration.cpp ir/vertex_declaration_registry.cpp ir/shader_program.cpp)
target_compile_features(t3render_ir PUBLIC cxx_std_17)
target_include_directories(t3render_ir PUBLIC ir)
if(MSVC)
  target_compile_options(t3render_ir PRIVATE /W4 /utf-8)
  set_property(TARGET t3render_ir PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(t3render_ir PRIVATE -Wall -Wextra)
endif()

if(MINGW)
  set(t3render_exports src/d3d8_mingw.def)
else()
  set(t3render_exports src/d3d8.def)
endif()
add_library(t3render SHARED
  src/proxy.cpp
  src/buffer_proxy.cpp
  src/vtable_layout.c
  src/capture.cpp
  ${t3render_exports})
target_compile_features(t3render PRIVATE cxx_std_17)
target_compile_definitions(t3render PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(t3render PRIVATE dxguid t3render_ir)
if(MSVC)
  target_compile_options(t3render PRIVATE /W4 /utf-8)
  set_property(TARGET t3render PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(t3render PRIVATE -Wall -Wextra)
  target_link_options(t3render PRIVATE -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(t3render PROPERTIES OUTPUT_NAME d3d8 PREFIX "")

if(T3RENDER_BUILD_EXPERIMENTAL_COMBINED_PROXY)
  set(t3render_minhook "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/third_party/minhook")
  add_library(t3render_experimental_combined_proxy SHARED
    src/proxy.cpp src/buffer_proxy.cpp src/vtable_layout.c src/capture.cpp src/frame_image.cpp
    src/scene_bridge_probe.cpp
    ${t3render_minhook}/src/buffer.c ${t3render_minhook}/src/hook.c
    ${t3render_minhook}/src/trampoline.c ${t3render_minhook}/src/hde/hde32.c ${t3render_exports})
  target_compile_features(t3render_experimental_combined_proxy PRIVATE cxx_std_17)
  target_include_directories(t3render_experimental_combined_proxy PRIVATE ${t3render_minhook}/include)
  target_compile_definitions(t3render_experimental_combined_proxy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
    T3RENDER_EXPERIMENTAL_COMBINED_PROXY)
  target_link_libraries(t3render_experimental_combined_proxy PRIVATE dxguid advapi32 t3render_ir)
  if(MSVC)
    target_compile_options(t3render_experimental_combined_proxy PRIVATE /W4 /utf-8)
    set_property(TARGET t3render_experimental_combined_proxy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(t3render_experimental_combined_proxy PRIVATE -Wall -Wextra)
    target_link_options(t3render_experimental_combined_proxy PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(t3render_experimental_combined_proxy PROPERTIES OUTPUT_NAME d3d8 PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/experimental_combined_proxy"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/experimental_combined_proxy"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/experimental_combined_proxy")

  # Synthetic target only: the bypass for a test function is never compiled
  # into the game proxy. Exercise the x86 hook's ABI before a live capture.

    add_executable(t3render_scene_bridge_hook_fixture EXCLUDE_FROM_ALL
      tests/scene_bridge_hook_fixture.cpp src/scene_bridge_probe.cpp
      ${t3render_minhook}/src/buffer.c ${t3render_minhook}/src/hook.c
      ${t3render_minhook}/src/trampoline.c ${t3render_minhook}/src/hde/hde32.c)
    target_compile_features(t3render_scene_bridge_hook_fixture PRIVATE cxx_std_17)
    target_compile_definitions(t3render_scene_bridge_hook_fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
      T3RENDER_SCENE_BRIDGE_TEST_FIXTURE)
    target_include_directories(t3render_scene_bridge_hook_fixture PRIVATE ${t3render_minhook}/include)
    target_link_libraries(t3render_scene_bridge_hook_fixture PRIVATE advapi32)
    target_compile_options(t3render_scene_bridge_hook_fixture PRIVATE -Wall -Wextra -msse2)
    target_link_options(t3render_scene_bridge_hook_fixture PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
